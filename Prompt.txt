<?php

namespace App\Services\Billing;

use App\Models\BillingCycle;
use App\Models\BillingTransaction;
use App\Models\RefundItem;
use App\Models\InvoiceLineItem;
use App\Models\InventoryItem;
use App\Models\Visit;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;
use Throwable;

/**
 * Billing Transaction Service
 *
 * Handles voiding transactions and processing refunds with full audit trails
 */
class BillingTransactionService
{
    /**
     * Void an entire billing cycle (complete void)
     *
     * @param int $billingCycleId
     * @param int $staffId
     * @param string $reason
     * @param int $facilityId
     * @return array
     */
    public function voidBillingCycle(int $billingCycleId, int $staffId, string $reason, int $facilityId): array
    {
        try {
            return DB::transaction(function () use ($billingCycleId, $staffId, $reason, $facilityId) {
                // Load billing cycle with relationships
                $billingCycle = BillingCycle::where('id', $billingCycleId)
                    ->where('facility_id', $facilityId)
                    ->with(['lineItems', 'transactions' => function($q) {
                        $q->where('status', 'completed');
                    }])
                    ->lockForUpdate()
                    ->first();

                if (!$billingCycle) {
                    return [
                        'success' => false,
                        'message' => 'Billing cycle not found.',
                        'errors' => ['billing_cycle_id' => ['Invalid billing cycle for this facility.']],
                    ];
                }

                // Check if already voided
                if ($billingCycle->billing_status === 'voided') {
                    return [
                        'success' => false,
                        'message' => 'This billing cycle has already been voided.',
                        'errors' => ['billing_cycle' => ['Cannot void an already voided billing cycle.']],
                    ];
                }

                // Check if too late to void (e.g., beyond policy period)
                if ($billingCycle->created_at->diffInDays(now()) > 30) {
                    return [
                        'success' => false,
                        'message' => 'Cannot void billing cycles older than 30 days. Please process a refund instead.',
                        'errors' => ['billing_cycle' => ['Billing cycle is too old to void.']],
                    ];
                }

                // Create void transaction record
                $voidTransaction = BillingTransaction::create([
                    'transaction_uuid' => Str::uuid(),
                    'billing_cycle_id' => $billingCycle->id,
                    'transaction_type' => 'void',
                    'amount' => $billingCycle->net_amount,
                    'payment_method_type' => 'system',
                    'payment_method_details' => json_encode(['void_type' => 'full_cycle']),
                    'reference_number' => "VOID-{$billingCycle->id}-" . time(),
                    'reason' => $reason,
                    'status' => 'completed',
                    'voided_at' => now(),
                    'voided_by_staff_id' => $staffId,
                    'void_reason' => $reason,
                    'metadata' => json_encode([
                        'original_billing_status' => $billingCycle->billing_status,
                        'original_net_amount' => $billingCycle->net_amount,
                        'original_payments' => [
                            'patient' => $billingCycle->patient_payment_received,
                            'insurance' => $billingCycle->insurance_payment_received,
                        ],
                    ]),
                    'created_by_staff_id' => $staffId,
                    'facility_id' => $facilityId,
                ]);

                // Restore inventory for inventory items
                $this->restoreInventoryForVoid($billingCycle->lineItems, $staffId);

                // Mark all line items as voided
                foreach ($billingCycle->lineItems as $lineItem) {
                    $lineItem->line_item_status = 'voided';
                    $lineItem->metadata = array_merge(
                        json_decode($lineItem->metadata ?? '{}', true) ?: [],
                        [
                            'voided_at' => now()->toISOString(),
                            'voided_by_staff_id' => $staffId,
                            'void_reason' => $reason,
                            'void_transaction_id' => $voidTransaction->id,
                        ]
                    );
                    $lineItem->save();
                }

                // Update billing cycle
                $billingCycle->billing_status = 'voided';
                $billingCycle->metadata = array_merge(
                    json_decode($billingCycle->metadata ?? '{}', true) ?: [],
                    [
                        'voided_at' => now()->toISOString(),
                        'voided_by_staff_id' => $staffId,
                        'void_reason' => $reason,
                        'void_transaction_id' => $voidTransaction->id,
                        'original_status' => $billingCycle->billing_status,
                    ]
                );
                $billingCycle->save();

                // Update visit status
                $this->updateVisitAfterVoid($billingCycle->visit_id, $staffId);

                Log::info('Billing cycle voided successfully', [
                    'billing_cycle_id' => $billingCycle->id,
                    'staff_id' => $staffId,
                    'void_transaction_id' => $voidTransaction->id,
                    'amount' => $billingCycle->net_amount,
                ]);

                return [
                    'success' => true,
                    'message' => 'Billing cycle voided successfully.',
                    'data' => [
                        'void_transaction_id' => $voidTransaction->id,
                        'void_transaction_uuid' => $voidTransaction->transaction_uuid,
                        'billing_cycle_id' => $billingCycle->id,
                        'amount_voided' => $billingCycle->net_amount,
                        'voided_at' => $voidTransaction->voided_at->toISOString(),
                    ],
                ];
            });
        } catch (Throwable $e) {
            Log::error('Failed to void billing cycle', [
                'billing_cycle_id' => $billingCycleId,
                'error' => $e->getMessage(),
                'trace' => $e->getTraceAsString(),
            ]);

            return [
                'success' => false,
                'message' => 'Failed to void billing cycle. All changes have been rolled back.',
                'errors' => ['system' => [$e->getMessage()]],
            ];
        }
    }

    /**
     * Process a refund (partial or full)
     *
     * @param array $data Refund data
     * @param int $staffId
     * @param int $facilityId
     * @return array
     */
    public function processRefund(array $data, int $staffId, int $facilityId): array
    {
        try {
            return DB::transaction(function () use ($data, $staffId, $facilityId) {
                // Load billing cycle
                $billingCycle = BillingCycle::where('id', $data['billing_cycle_id'])
                    ->where('facility_id', $facilityId)
                    ->with(['lineItems'])
                    ->lockForUpdate()
                    ->first();

                if (!$billingCycle) {
                    return [
                        'success' => false,
                        'message' => 'Billing cycle not found.',
                    ];
                }

                // Calculate refund totals and validate items
                $refundValidation = $this->validateRefundItems($data['items'], $billingCycle);
                if (!$refundValidation['success']) {
                    return $refundValidation;
                }

                // Create refund transaction
                $refundTransaction = BillingTransaction::create([
                    'transaction_uuid' => Str::uuid(),
                    'billing_cycle_id' => $billingCycle->id,
                    'transaction_type' => 'refund',
                    'amount' => $refundValidation['total_refund'],
                    'payment_method_type' => $data['refund_method'] ?? 'original',
                    'payment_method_details' => json_encode($data['refund_details'] ?? []),
                    'reference_number' => $data['reference_number'] ?? "REF-{$billingCycle->id}-" . time(),
                    'reason' => $data['reason'],
                    'status' => 'completed',
                    'metadata' => json_encode([
                        'refund_type' => $data['type'] ?? 'partial', // 'full' or 'partial'
                        'original_billing_status' => $billingCycle->billing_status,
                    ]),
                    'created_by_staff_id' => $staffId,
                    'facility_id' => $facilityId,
                ]);

                // Create refund items and process each
                $refundedItems = [];
                foreach ($data['items'] as $itemData) {
                    $lineItem = InvoiceLineItem::find($itemData['line_item_id']);
                    
                    // Create refund item record
                    $refundItem = RefundItem::create([
                        'refund_transaction_id' => $refundTransaction->id,
                        'line_item_id' => $lineItem->id,
                        'quantity_refunded' => $itemData['quantity'],
                        'unit_price_at_time' => $lineItem->unit_price_at_time,
                        'subtotal_refunded' => $itemData['subtotal'] ?? ($lineItem->unit_price_at_time * $itemData['quantity']),
                        'discount_refunded' => $itemData['discount'] ?? 0,
                        'tax_refunded' => $itemData['tax'] ?? 0,
                        'total_refunded' => $itemData['total'],
                        'reason' => $itemData['reason'] ?? $data['reason'],
                        'requires_restock' => $itemData['requires_restock'] ?? false,
                        'metadata' => json_encode($itemData['metadata'] ?? []),
                    ]);

                    // Restock inventory if required
                    if ($refundItem->requires_restock) {
                        $this->restockInventoryItem($lineItem, $itemData['quantity'], $staffId);
                        
                        $refundItem->restocked_at = now();
                        $refundItem->restocked_by_staff_id = $staffId;
                        $refundItem->save();
                    }

                    // Update line item with refund information
                    $lineItemMetadata = json_decode($lineItem->metadata ?? '{}', true) ?: [];
                    $lineItemMetadata['refunds'][] = [
                        'refund_transaction_id' => $refundTransaction->id,
                        'refund_item_id' => $refundItem->id,
                        'quantity_refunded' => $itemData['quantity'],
                        'refunded_at' => now()->toISOString(),
                        'refunded_by_staff_id' => $staffId,
                        'reason' => $itemData['reason'] ?? $data['reason'],
                    ];
                    
                    // Update line item status if fully refunded
                    $totalRefundedQuantity = collect($lineItemMetadata['refunds'] ?? [])->sum('quantity_refunded');
                    if ($totalRefundedQuantity >= $lineItem->quantity) {
                        $lineItem->line_item_status = 'refunded';
                    } elseif ($totalRefundedQuantity > 0) {
                        $lineItem->line_item_status = 'partially_refunded';
                    }
                    
                    $lineItem->metadata = $lineItemMetadata;
                    $lineItem->save();

                    $refundedItems[] = $refundItem;
                }

                // Update billing cycle with refund amounts
                $this->updateBillingCycleAfterRefund($billingCycle, $refundValidation, $refundTransaction, $staffId);

                Log::info('Refund processed successfully', [
                    'billing_cycle_id' => $billingCycle->id,
                    'refund_transaction_id' => $refundTransaction->id,
                    'amount' => $refundValidation['total_refund'],
                    'items_refunded' => count($refundedItems),
                    'staff_id' => $staffId,
                ]);

                return [
                    'success' => true,
                    'message' => 'Refund processed successfully.',
                    'data' => [
                        'refund_transaction_id' => $refundTransaction->id,
                        'refund_transaction_uuid' => $refundTransaction->transaction_uuid,
                        'reference_number' => $refundTransaction->reference_number,
                        'amount_refunded' => $refundValidation['total_refund'],
                        'items_refunded' => $refundedItems,
                        'refunded_at' => $refundTransaction->created_at->toISOString(),
                    ],
                ];
            });
        } catch (Throwable $e) {
            Log::error('Failed to process refund', [
                'billing_cycle_id' => $data['billing_cycle_id'] ?? null,
                'error' => $e->getMessage(),
                'trace' => $e->getTraceAsString(),
            ]);

            return [
                'success' => false,
                'message' => 'Failed to process refund. All changes have been rolled back.',
                'errors' => ['system' => [$e->getMessage()]],
            ];
        }
    }

    /**
     * Validate refund items
     *
     * @param array $items
     * @param BillingCycle $billingCycle
     * @return array
     */
    protected function validateRefundItems(array $items, BillingCycle $billingCycle): array
    {
        $totalRefund = 0;
        $validatedItems = [];

        foreach ($items as $item) {
            $lineItem = $billingCycle->lineItems->firstWhere('id', $item['line_item_id']);
            
            if (!$lineItem) {
                return [
                    'success' => false,
                    'message' => "Line item ID {$item['line_item_id']} not found in this billing cycle.",
                ];
            }

            // Check if already refunded
            $metadata = json_decode($lineItem->metadata ?? '{}', true) ?: [];
            $alreadyRefunded = collect($metadata['refunds'] ?? [])->sum('quantity_refunded');
            $availableToRefund = $lineItem->quantity - $alreadyRefunded;

            if ($item['quantity'] > $availableToRefund) {
                return [
                    'success' => false,
                    'message' => "Cannot refund {$item['quantity']} of item '{$lineItem->service_description}'. Only {$availableToRefund} available to refund.",
                ];
            }

            // Calculate proportional amounts
            $refundRatio = $item['quantity'] / $lineItem->quantity;
            $subtotalRefund = $lineItem->line_total_amount * $refundRatio;
            $discountRefund = ($lineItem->discount_amount ?? 0) * $refundRatio;
            $taxRefund = 0; // Calculate based on your tax logic
            $totalRefundItem = $subtotalRefund - $discountRefund + $taxRefund;

            $validatedItems[] = [
                'line_item_id' => $lineItem->id,
                'quantity' => $item['quantity'],
                'subtotal' => $subtotalRefund,
                'discount' => $discountRefund,
                'tax' => $taxRefund,
                'total' => $totalRefundItem,
                'reason' => $item['reason'] ?? null,
                'requires_restock' => $item['requires_restock'] ?? false,
            ];

            $totalRefund += $totalRefundItem;
        }

        return [
            'success' => true,
            'items' => $validatedItems,
            'total_refund' => $totalRefund,
        ];
    }

    /**
     * Restore inventory for voided billing cycle
     *
     * @param \Illuminate\Support\Collection $lineItems
     * @param int $staffId
     * @return void
     */
    protected function restoreInventoryForVoid($lineItems, int $staffId): void
    {
        foreach ($lineItems as $lineItem) {
            $serviceSnapshot = json_decode($lineItem->service_version_snapshot ?? '{}', true);
            
            $inventoryItem = InventoryItem::where('item_code', $serviceSnapshot['code'] ?? $lineItem->service_code)
                ->lockForUpdate()
                ->first();

            if (!$inventoryItem) {
                continue;
            }

            $previousQuantity = $inventoryItem->package_quantity;
            $inventoryItem->package_quantity += $lineItem->quantity;
            $inventoryItem->updated_by_staff_id = $staffId;

            $metadata = json_decode($inventoryItem->metadata ?? '{}', true) ?: [];
            $metadata['stock_restorations'][] = [
                'restored_at' => now()->toISOString(),
                'restored_by_staff_id' => $staffId,
                'units_restored' => $lineItem->quantity,
                'previous_quantity' => $previousQuantity,
                'new_quantity' => $inventoryItem->package_quantity,
                'reason' => 'billing_cycle_voided',
                'billing_cycle_id' => $lineItem->billing_cycle_id,
                'line_item_id' => $lineItem->id,
            ];

            $inventoryItem->metadata = $metadata;
            $inventoryItem->save();

            Log::info('Inventory restored for voided billing', [
                'item_code' => $inventoryItem->item_code,
                'units_restored' => $lineItem->quantity,
                'billing_cycle_id' => $lineItem->billing_cycle_id,
            ]);
        }
    }

    /**
     * Restock inventory for refunded item
     *
     * @param InvoiceLineItem $lineItem
     * @param float $quantity
     * @param int $staffId
     * @return void
     */
    protected function restockInventoryItem(InvoiceLineItem $lineItem, float $quantity, int $staffId): void
    {
        $serviceSnapshot = json_decode($lineItem->service_version_snapshot ?? '{}', true);
        
        $inventoryItem = InventoryItem::where('item_code', $serviceSnapshot['code'] ?? $lineItem->service_code)
            ->lockForUpdate()
            ->first();

        if (!$inventoryItem) {
            Log::warning('Attempted to restock non-inventory item', [
                'line_item_id' => $lineItem->id,
                'service_code' => $serviceSnapshot['code'] ?? $lineItem->service_code,
            ]);
            return;
        }

        $previousQuantity = $inventoryItem->package_quantity;
        $inventoryItem->package_quantity += $quantity;
        $inventoryItem->updated_by_staff_id = $staffId;

        $metadata = json_decode($inventoryItem->metadata ?? '{}', true) ?: [];
        $metadata['stock_restorations'][] = [
            'restored_at' => now()->toISOString(),
            'restored_by_staff_id' => $staffId,
            'units_restored' => $quantity,
            'previous_quantity' => $previousQuantity,
            'new_quantity' => $inventoryItem->package_quantity,
            'reason' => 'refund',
            'line_item_id' => $lineItem->id,
        ];

        $inventoryItem->metadata = $metadata;
        $inventoryItem->save();
    }

    /**
     * Update billing cycle after refund
     *
     * @param BillingCycle $billingCycle
     * @param array $refundValidation
     * @param BillingTransaction $refundTransaction
     * @param int $staffId
     * @return void
     */
    protected function updateBillingCycleAfterRefund(
        BillingCycle $billingCycle,
        array $refundValidation,
        BillingTransaction $refundTransaction,
        int $staffId
    ): void {
        // Recalculate totals
        $newPatientPayment = max(0, $billingCycle->patient_payment_received - $refundValidation['total_refund']);
        $newNetAmount = $billingCycle->net_amount - $refundValidation['total_refund'];

        // Determine new status
        if ($newPatientPayment <= 0) {
            $newStatus = 'refunded';
        } elseif ($newPatientPayment < $billingCycle->patient_responsibility_amount) {
            $newStatus = 'partially_refunded';
        } else {
            $newStatus = $billingCycle->billing_status; // Keep original if fully paid still
        }

        // Update billing cycle
        $billingCycle->patient_payment_received = $newPatientPayment;
        $billingCycle->net_amount = $newNetAmount;
        $billingCycle->billing_status = $newStatus;
        
        $metadata = json_decode($billingCycle->metadata ?? '{}', true) ?: [];
        $metadata['refunds'][] = [
            'refund_transaction_id' => $refundTransaction->id,
            'refund_amount' => $refundValidation['total_refund'],
            'refunded_at' => now()->toISOString(),
            'refunded_by_staff_id' => $staffId,
            'reason' => $refundTransaction->reason,
        ];
        
        $billingCycle->metadata = $metadata;
        $billingCycle->save();

        // Update visit
        $visit = Visit::find($billingCycle->visit_id);
        if ($visit) {
            $visit->patient_estimated_responsibility = $newPatientPayment;
            $visit->payment_status = $newPatientPayment <= 0 ? 'refunded' : 
                ($newPatientPayment < $billingCycle->patient_responsibility_amount ? 'partially_refunded' : $visit->payment_status);
            $visit->save();
        }
    }

    /**
     * Update visit after void
     *
     * @param int $visitId
     * @param int $staffId
     * @return void
     */
    protected function updateVisitAfterVoid(int $visitId, int $staffId): void
    {
        $visit = Visit::find($visitId);
        if ($visit) {
            $visit->current_phase = 'billing';
            $visit->payment_status = 'voided';
            $visit->estimated_total_charges = 0;
            $visit->patient_estimated_responsibility = 0;
            
            $metadata = json_decode($visit->metadata ?? '{}', true) ?: [];
            $metadata['voided_billing'][] = [
                'voided_at' => now()->toISOString(),
                'voided_by_staff_id' => $staffId,
            ];
            $visit->metadata = $metadata;
            
            $visit->save();
        }
    }

    /**
     * Get transaction history for a billing cycle
     *
     * @param int $billingCycleId
     * @return array
     */
    public function getTransactionHistory(int $billingCycleId): array
    {
        $transactions = BillingTransaction::where('billing_cycle_id', $billingCycleId)
            ->with(['createdByStaff.user', 'voidedByStaff.user', 'refunds'])
            ->orderBy('created_at', 'desc')
            ->get();

        return [
            'success' => true,
            'data' => $transactions->map(function ($transaction) {
                return [
                    'id' => $transaction->id,
                    'uuid' => $transaction->transaction_uuid,
                    'type' => $transaction->transaction_type,
                    'amount' => $transaction->amount,
                    'payment_method' => $transaction->payment_method_type,
                    'reference' => $transaction->reference_number,
                    'reason' => $transaction->reason,
                    'status' => $transaction->status,
                    'created_at' => $transaction->created_at->toISOString(),
                    'created_by' => $transaction->createdByStaff?->user?->display_name,
                    'voided_at' => $transaction->voided_at?->toISOString(),
                    'voided_by' => $transaction->voidedByStaff?->user?->display_name,
                    'void_reason' => $transaction->void_reason,
                ];
            }),
        ];
    }
}
